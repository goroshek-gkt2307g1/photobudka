<script>
(function () {
  var form = document.getElementById('vb-h4p-form');
  if (!form) return;

  var pass = document.getElementById('vb-h4p-pass');
  var pass2 = document.getElementById('vb-h4p-pass2');
  var pass2Error = document.getElementById('vb-h4p-pass2-error');
  var btn = form.querySelector('[type="submit"]');
  var eyes = form.querySelectorAll('.vb-h4p-eye');

  Array.prototype.forEach.call(eyes, function (eye) {
    eye.addEventListener('click', function () {
      var input = document.getElementById(eye.getAttribute('data-eye'));
      if (!input) return;
      var show = input.type === 'password';
      input.type = show ? 'text' : 'password';
      eye.textContent = show ? 'Скрыть' : 'Показать';
      eye.setAttribute('aria-label', show ? 'Скрыть пароль' : 'Показать пароль');
    });
  });

  function checkMatch(showError) {
    if (!pass || !pass2 || !pass2Error) return true;
    var mismatch = pass.value !== pass2.value;
    if (mismatch && showError) {
      pass2Error.textContent = 'Пароли не совпадают';
    } else {
      pass2Error.textContent = '';
    }
    return !mismatch;
  }

  if (pass2) {
    pass2.addEventListener('input', function () { checkMatch(false); });
    pass2.addEventListener('blur', function () { checkMatch(true); });
  }

  form.addEventListener('submit', function (event) {
    event.preventDefault();
    if (!checkMatch(true)) {
      pass2.focus();
      return;
    }
    if (typeof window.t_forms__initBtnClick !== 'function') return;
    window.t_forms__initBtnClick({
      target: btn,
      srcElement: btn,
      preventDefault: function () {}
    });
  });
})();
</script>

<script>
(function () {
  var root = document.getElementById('vb-k7m-root');
  if (!root) return;
  var each = function (list, fn) { Array.prototype.forEach.call(list, fn); };

  /* Показать / скрыть пароль */
  each(root.querySelectorAll('[data-vb-eye]'), function (btn) {
    btn.addEventListener('click', function () {
      var input = document.getElementById(btn.getAttribute('data-vb-eye'));
      if (!input) return;
      var show = input.type === 'password';
      input.type = show ? 'text' : 'password';
      btn.textContent = show ? 'Скрыть' : 'Показать';
      btn.setAttribute('aria-label', show ? 'Скрыть пароль' : 'Показать пароль');
    });
  });
  each(root.querySelectorAll('[data-vb-eye-icon]'), function (btn) {
    btn.addEventListener('click', function () {
      var input = btn.parentNode ? btn.parentNode.querySelector('input') : null;
      if (!input) return;
      var show = input.type === 'password';
      input.type = show ? 'text' : 'password';
      btn.setAttribute('aria-label', show ? 'Скрыть пароль' : 'Показать пароль');
    });
  });

  /* Нижняя навигация — переключение активного таба */
  each(root.querySelectorAll('.vb-k7m-tabs'), function (bar) {
    var items = bar.querySelectorAll('.vb-k7m-tab');
    each(items, function (item) {
      item.addEventListener('click', function () {
        each(items, function (i) { i.classList.remove('is-active'); });
        item.classList.add('is-active');
      });
    });
  });

  /* Сегмент «Вход / Регистрация» */
  each(root.querySelectorAll('.vb-k7m-seg'), function (seg) {
    var btns = seg.querySelectorAll('[data-vb-seg]');
    each(btns, function (b) {
      b.addEventListener('click', function () {
        each(btns, function (i) { i.classList.remove('is-active'); });
        b.classList.add('is-active');
      });
    });
  });

  /* Описание: показать полностью */
  each(root.querySelectorAll('[data-vb-desc]'), function (btn) {
    btn.addEventListener('click', function () {
      var text = document.getElementById('vb-k7m-desc');
      if (!text) return;
      var open = text.classList.toggle('is-open');
      btn.textContent = open ? 'Свернуть' : 'Показать полностью';
      btn.setAttribute('aria-expanded', open ? 'true' : 'false');
    });
  });

  /* Аккордеон «Мои заявки» */
  each(root.querySelectorAll('[data-vb-acc]'), function (btn) {
    btn.addEventListener('click', function () {
      var panel = document.getElementById(btn.getAttribute('aria-controls'));
      if (!panel) return;
      var open = btn.getAttribute('aria-expanded') === 'true';
      btn.setAttribute('aria-expanded', open ? 'false' : 'true');
      panel.hidden = open;
    });
  });

  /* Приватный проект → поле пароля */
  each(root.querySelectorAll('[data-vb-priv]'), function (box) {
    box.addEventListener('change', function () {
      var group = document.getElementById('vb-k7m-p-passgroup');
      if (!group) return;
      group.hidden = !box.checked;
      if (!box.checked) {
        var input = document.getElementById('vb-k7m-p-pass');
        if (input) input.value = '';
      }
    });
  });

  /* Выбор фото в альбоме */
  var grid = document.getElementById('vb-k7m-grid');
  var bar = document.getElementById('vb-k7m-bar');
  var countEl = document.getElementById('vb-k7m-bar-count');
  var del = document.getElementById('vb-k7m-del');
  var syncPhotos = function () {
    if (!grid || !bar || !countEl) return;
    var checked = grid.querySelectorAll('.vb-k7m-tile__check:checked');
    countEl.textContent = 'Выбрано фото: ' + checked.length;
    bar.hidden = checked.length === 0;
    each(grid.querySelectorAll('.vb-k7m-tile'), function (tile) {
      var box = tile.querySelector('.vb-k7m-tile__check');
      tile.classList.toggle('is-checked', !!(box && box.checked));
    });
  };
  if (grid) {
    grid.addEventListener('change', function (e) {
      if (e.target && e.target.classList && e.target.classList.contains('vb-k7m-tile__check')) syncPhotos();
    });
  }
  if (del && grid) {
    del.addEventListener('click', function () {
      each(grid.querySelectorAll('.vb-k7m-tile'), function (tile) {
        var box = tile.querySelector('.vb-k7m-tile__check');
        if (box && box.checked && tile.parentNode) tile.parentNode.removeChild(tile);
      });
      syncPhotos();
    });
  }
  syncPhotos();

  /* Свайп влево по карточке заявки */
  var OPEN = -96;
  each(root.querySelectorAll('[data-vb-swipe]'), function (card) {
    var startX = 0;
    var base = 0;
    var current = 0;
    var active = false;
    card.addEventListener('pointerdown', function (e) {
      active = true;
      startX = e.clientX;
      base = current;
      card.classList.add('is-dragging');
    });
    card.addEventListener('pointermove', function (e) {
      if (!active) return;
      var dx = e.clientX - startX;
      var next = Math.min(0, Math.max(OPEN, base + dx));
      current = next;
      card.style.transform = 'translateX(' + next + 'px)';
    });
    var end = function () {
      if (!active) return;
      active = false;
      card.classList.remove('is-dragging');
      current = current < OPEN * 0.4 ? OPEN : 0;
      card.style.transform = 'translateX(' + current + 'px)';
    };
    card.addEventListener('pointerup', end);
    card.addEventListener('pointercancel', end);
    card.addEventListener('pointerleave', end);
  });
})();
