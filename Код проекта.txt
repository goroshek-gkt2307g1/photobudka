let db;

const initDb = async () => {
    if (!db) {
        db = await open({
            filename: 'C:/Users/goroshek/Desktop/колледж/4 курс/sqllite/Photobooth.db',
            driver: sqlite3.Database
        });
    }

    // Включаем поддержку внешних ключей
    await db.exec('PRAGMA foreign_keys = ON;');

    // Роли
    await db.exec(`
        CREATE TABLE IF NOT EXISTS Roles (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL
        )`);

    // Пользователи
    await db.exec(`
        CREATE TABLE IF NOT EXISTS Users (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            email TEXT NOT NULL UNIQUE,
            password_hash TEXT NOT NULL,
            role_id_FK INTEGER NOT NULL,
            name TEXT NOT NULL,
            is_active INTEGER NOT NULL DEFAULT 1,
            created_at TEXT NOT NULL DEFAULT (datetime('now')),
            updated_at TEXT,
            FOREIGN KEY (role_id_FK) REFERENCES Roles(id)
        )`);

    // Профили фотографов
    await db.exec(`
        CREATE TABLE IF NOT EXISTS Profiles (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            user_id_FK INTEGER NOT NULL UNIQUE,
            city TEXT,
            genres TEXT,
            price_hour INTEGER,
            description TEXT,
            avatar_url TEXT,
            FOREIGN KEY (user_id_FK) REFERENCES Users(id) ON DELETE CASCADE
        )`);

    // Проекты
    await db.exec(`
        CREATE TABLE IF NOT EXISTS Projects (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            profile_id_FK INTEGER NOT NULL,
            title TEXT NOT NULL,
            description TEXT,
            genre TEXT,
            shoot_date TEXT,
            is_private INTEGER NOT NULL DEFAULT 0,
            access_password TEXT,
            cover_photo_url TEXT,
            created_at TEXT NOT NULL DEFAULT (datetime('now')),
            FOREIGN KEY (profile_id_FK) REFERENCES Profiles(id) ON DELETE CASCADE
        )`);

    // Фотографии
    await db.exec(`
        CREATE TABLE IF NOT EXISTS Photos (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            project_id_FK INTEGER NOT NULL,
            file_path TEXT NOT NULL,
            uploaded_at TEXT NOT NULL DEFAULT (datetime('now')),
            FOREIGN KEY (project_id_FK) REFERENCES Projects(id) ON DELETE CASCADE
        )`);

    // Заявки
    await db.exec(`
        CREATE TABLE IF NOT EXISTS Requests (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            customer_id_FK INTEGER,
            photographer_id_FK INTEGER NOT NULL,
            name TEXT NOT NULL,
            contact TEXT NOT NULL,
            desired_date TEXT,
            genre TEXT,
            comment TEXT,
            status TEXT NOT NULL DEFAULT 'new',
            created_at TEXT NOT NULL DEFAULT (datetime('now')),
            FOREIGN KEY (customer_id_FK) REFERENCES Users(id),
            FOREIGN KEY (photographer_id_FK) REFERENCES Profiles(id)
        )`);

    // Стартовые роли (если их ещё нет)
    const rolesCount = await db.get('SELECT COUNT(*) as count FROM Roles');
    if (rolesCount.count === 0) {
        await db.run('INSERT INTO Roles (name) VALUES (?)', ['photographer']);
        await db.run('INSERT INTO Roles (name) VALUES (?)', ['customer']);
    }
};

             VALUES (?, ?, ?, ?, ?, ?)`,

        );

        SELECT p.*, u.name AS photographer_name 
        FROM ${TABLE_NAME} p 
        JOIN Users u ON p.user_id_FK = u.id`
    ),
    getProfileById: async (id) => await getDb().get(
        `SELECT * FROM ${TABLE_NAME} WHERE id = ?`, id
    ),
    getProfileByUserId: async (userId) => await getDb().get(
        `SELECT * FROM ${TABLE_NAME} WHERE user_id_FK = ?`, userId
    ),
    updateProfile: async (id, data) => {
        await getDb().run(
            `UPDATE ${TABLE_NAME} 
             SET city = ?, genres = ?, price_hour = ?, description = ?, avatar_url = ? 
             WHERE id = ?`,
            data.city, data.genres, data.price_hour, data.description, data.avatar_url, id
        );
    },
    searchProfiles: async (city, genre, maxPrice) => {
        let query = `SELECT p.*, u.name AS photographer_name 
                     FROM ${TABLE_NAME} p 
                     JOIN Users u ON p.user_id_FK = u.id 
                     WHERE 1=1`;
        const params = [];
        if (city) { query += ` AND p.city = ?`; params.push(city); }
        if (genre) { query += ` AND p.genres LIKE ?`; params.push(`%${genre}%`); }
        if (maxPrice) { query += ` AND p.price_hour <= ?`; params.push(maxPrice); }
        return await getDb().all(query, ...params);
    },
    deleteProfile: async (id) => {
        await getDb().run(`DELETE FROM ${TABLE_NAME} WHERE id = ?`, id);
    },

const jwt = require('jsonwebtoken')
const JWT_SECRET = process.env.JWT_SECRET || 'lolkek'
const COOKIE_NAME = 'token'

const authMiddleware = (req, res, next) => {
    const token = req.cookies[COOKIE_NAME];

    if (!token) {
        return res.status(401).json({
            message: 'Пользователь не авторизован'
        });
    }

    try {
        const decoded = jwt.verify(token, JWT_SECRET);
        req.userId = decoded.userId;
        req.userRole = decoded.role;
        next();
    } catch (err) {
        return res.status(401).json({
            message: 'Недействительный токен'
        });
    }
};

module.exports = authMiddleware;



const express = require('express');
const jwt = require('jsonwebtoken');
const bcrypt = require('bcrypt');
const authRouter = express.Router();
const { getUserByEmail } = require('../db/users');

const JWT_SECRET = process.env.JWT_SECRET || 'lolkek';
const COOKIE_NAME = 'token';

authRouter.post('/', async (req, res) => {
    const user = await getUserByEmail(req.body.email);

    if (!user) {
        return res.status(404).json({
            message: 'Такой пользователь не найден'
        });
    }

    const isValid = await bcrypt.compare(req.body.password, user.password_hash);

    if (!isValid) {
        return res.status(400).json({
            message: 'Пароль неверный'
        });
    }

    const token = jwt.sign(
        { userId: user.id, role: user.role_id_FK },
        JWT_SECRET,
        { expiresIn: '24h' }
    );

    res.cookie(COOKIE_NAME, token, {
        maxAge: 24 * 60 * 60 * 1000,
        httpOnly: true,
        sameSite: process.env.NODE_ENV === 'production' ? 'none' : 'lax',
        secure: process.env.NODE_ENV === 'production'
    });

    res.status(200).json({ ok: true });
});

authRouter.delete('/', async (req, res) => {
    res.clearCookie(COOKIE_NAME);
    res.status(200).json({ ok: true });
});

module.exports = authRouter;

import { ref, computed } from 'vue';

interface User {
    id: number;
    email: string;
    name: string;
    role_id_FK: number;

export const useAuthStore = defineStore('auth', () => {
    const user = ref<User | null>(null);
    const loading = ref(false);
    const error = ref('');

    const isAuthenticated = computed(() => !!user.value);
    const isPhotographer = computed(() => user.value?.role_id_FK === 1);
    const isCustomer = computed(() => user.value?.role_id_FK === 2);

    async function login(email: string, password: string) {
        loading.value = true;
        error.value = '';
        try {
            await API.auth.login({ email, password });
            await fetchCurrentUser();
        } catch (e) {
            error.value = (e as Error).message;
            throw e;
        } finally {
            loading.value = false;
        }
    }

    async function register(data: {
        email: string;
        password: string;
        name: string;
        role: 'photographer' | 'customer';
    }) {
        loading.value = true;
        error.value = '';
        try {
            await API.users.register(data);
            await login(data.email, data.password);
        } catch (e) {
            error.value = (e as Error).message;
            throw e;
        } finally {
            loading.value = false;
        }
    }

    async function fetchCurrentUser() {
        try {
            user.value = await API.users.getCurrentUser();
        } catch {
            user.value = null;
        }
    }

    async function logout() {
        await API.auth.logout();
        user.value = null;
    }

    return {
        user,
        loading,
        error,
        isAuthenticated,
        isPhotographer,
        isCustomer,
        login,
        register,
        fetchCurrentUser,
        logout
    };
});

<script setup lang="ts">
import { ref, onMounted, onUnmounted, computed } from 'vue';
import { useRoute } from 'vue-router';
import { API } from '@/services/api';
import { useAuthStore } from '@/stores/auth';

const route = useRoute();
const auth = useAuthStore();
const projectId = Number(route.params.id);

const project = ref<any>(null);
const photos = ref<any[]>([]);
const loading = ref(true);
const error = ref('');

const password = ref('');
const isUnlocked = ref(false);

// Редактирование
const isEditing = ref(false);
const editForm = ref({ title: '', description: '', genre: '', shoot_date: '', is_private: false, access_password: '' });

// Выделение фото
const selectedPhotoIds = ref<number[]>([]);
const showDeleteModal = ref(false);

// Просмотр фото (Lightbox)
const activePhotoIndex = ref<number | null>(null);
const activePhoto = computed(() => activePhotoIndex.value !== null ? photos.value[activePhotoIndex.value] : null);

const isOwner = computed(() => {
    if (!auth.user || !project.value) return false;
    return auth.user.id === project.value.user_id_FK;
});

const canDownload = computed(() => {
    if (!project.value) return false;
    if (isOwner.value) return true;
    if (project.value.is_private === 1 && isUnlocked.value) return true;
    return false;
});

async function loadProject() {
    loading.value = true;
    try {
        project.value = await API.projects.getById(projectId);
        editForm.value = {
            title: project.value.title,
            description: project.value.description || '',
            genre: project.value.genre || '',
            shoot_date: project.value.shoot_date ? project.value.shoot_date.split('T')[0] : '',
            is_private: project.value.is_private === 1,
            access_password: project.value.access_password || ''
        };
        if (!project.value.is_private || isUnlocked.value || isOwner.value) {
            photos.value = await API.photos.getByProject(projectId);
        }
    } catch (e) {
        error.value = 'Проект не найден';
    } finally {
        loading.value = false;
    }
}

async function handleUnlock() {
    try {
        await API.projects.accessPrivate(projectId, password.value);
        isUnlocked.value = true;
        await loadProject();
    } catch (e) { alert('Неверный пароль'); }
}

async function saveProject() {
    try {
        await API.projects.update(projectId, {
            ...editForm.value,
            access_password: editForm.value.is_private ? editForm.value.access_password : undefined
        });
        isEditing.value = false;
        await loadProject();
    } catch (e) { alert('Ошибка сохранения'); }
}

async function handleUpload(event: Event) {
    const target = event.target as HTMLInputElement;
    if (!target.files || target.files.length === 0) return;
    try {
        loading.value = true;
        await API.photos.upload(projectId, target.files);
        await loadProject();
    } catch (e) { alert('Ошибка загрузки'); } finally { loading.value = false; target.value = ''; }
}

function togglePhotoSelection(id: number) {
    if (selectedPhotoIds.value.includes(id)) {
        selectedPhotoIds.value = selectedPhotoIds.value.filter(pid => pid !== id);
    } else {
        selectedPhotoIds.value.push(id);
    }
}

async function confirmDelete() {
    try {
        for (const id of selectedPhotoIds.value) await API.photos.delete(id);
        selectedPhotoIds.value = [];
        showDeleteModal.value = false;
        await loadProject();
    } catch (e) { alert('Ошибка удаления'); }
}

// Lightbox навигация
function openPhoto(index: number) { activePhotoIndex.value = index; }
function closePhoto() { activePhotoIndex.value = null; }
function nextPhoto() { if (activePhotoIndex.value !== null && activePhotoIndex.value < photos.value.length - 1) activePhotoIndex.value++; }
function prevPhoto() { if (activePhotoIndex.value !== null && activePhotoIndex.value > 0) activePhotoIndex.value--; }

function handleKeydown(e: KeyboardEvent) {
    if (activePhotoIndex.value === null) return;
    if (e.key === 'ArrowRight') nextPhoto();
    if (e.key === 'ArrowLeft') prevPhoto();
    if (e.key === 'Escape') closePhoto();
}

onMounted(() => {
    window.addEventListener('keydown', handleKeydown);
    loadProject();
});
onUnmounted(() => window.removeEventListener('keydown', handleKeydown));
</script>

import { ref, onMounted } from 'vue';
import { useRoute, useRouter } from 'vue-router';
import { API } from '@/services/api';
import { useAuthStore } from '@/stores/auth';

const route = useRoute();
const router = useRouter();
const auth = useAuthStore();
const photographerId = Number(route.params.id);

const profile = ref<any>(null);
const loading = ref(true);
const error = ref('');

const name = ref(auth.user?.name || '');
const contact = ref(auth.user?.email || '');
const desiredDate = ref('');
const genre = ref('');
const comment = ref('');
const isSending = ref(false);

onMounted(async () => {
    try {
        const allProfiles = await API.profiles.getAll();
        profile.value = allProfiles.find((p: any) => p.user_id_FK === photographerId);
        if (!profile.value) error.value = 'Фотограф не найден';
    } finally { loading.value = false; }
});

async function submitRequest() {
    isSending.value = true;
    try {
        await API.requests.create({
            photographer_id: profile.value.id,
            name: name.value,
            contact: contact.value,
            desired_date: desiredDate.value,
            genre: genre.value,
            comment: comment.value
        });
        alert('Заявка отправлена!');
        router.push('/dashboard');
    } catch (e) { error.value = (e as Error).message; } finally { isSending.value = false; }
}
</script>
