  each(root.querySelectorAll('.r8k-side'), function (side) {
    var items = side.querySelectorAll('.r8k-nav__i:not(.r8k-nav__i--danger)');
    each(items, function (item) {
      item.addEventListener('click', function () {
        each(items, function (i) { i.classList.remove('is-active'); });
        item.classList.add('is-active');
      });
    });
  });

  /* Сброс фильтров */
  each(root.querySelectorAll('[data-r8k-reset]'), function (btn) {
    btn.addEventListener('click', function () {
      var scope = btn.parentNode;
      if (!scope) return;
      each(scope.querySelectorAll('select'), function (sel) { sel.selectedIndex = 0; });
      each(scope.querySelectorAll('input[type="search"], input[type="text"]'), function (inp) { inp.value = ''; });
    });
  });

  /* Показать / скрыть пароль */
  each(root.querySelectorAll('[data-r8k-eye]'), function (btn) {
    btn.addEventListener('click', function () {
      var input = document.getElementById(btn.getAttribute('data-r8k-eye'));
      if (!input) return;
      var show = input.type === 'password';
      input.type = show ? 'text' : 'password';
      btn.textContent = show ? 'Скрыть' : 'Показать';
    });
  });

  /* Стр. 5: таблица заявок и модалка */

  if (table && overlay) {
    var m = {
      name: document.getElementById('r8k-m-name'),
      contact: document.getElementById('r8k-m-contact'),
      date: document.getElementById('r8k-m-date'),
      genre: document.getElementById('r8k-m-genre'),
      note: document.getElementById('r8k-m-note'),
      status: document.getElementById('r8k-m-status'),
      badge: document.getElementById('r8k-modal-badge')
    };
    var currentRow = null;

    var setStatus = function (row, kind) {
      var cell = row.querySelector('td:nth-child(6)');
      var acts = row.querySelector('.r8k-table__acts');
      if (!cell || !acts) return;
      cell.innerHTML = '';
      var badge = document.createElement('span');
      if (kind === 'ok') {
        badge.className = 'r8k-badge r8k-badge--ok';
        badge.textContent = 'Принята';
        if (m.badge) { m.badge.className = 'r8k-badge r8k-badge--ok'; m.badge.textContent = 'Принята'; }
        if (m.status) m.status.textContent = 'Принята';
      } else {
        badge.className = 'r8k-badge r8k-badge--no';
        badge.textContent = 'Отклонена';
        if (m.badge) { m.badge.className = 'r8k-badge r8k-badge--no'; m.badge.textContent = 'Отклонена'; }
        if (m.status) m.status.textContent = 'Отклонена';
      }
      cell.appendChild(badge);
      acts.innerHTML = '';
      var info = document.createElement('button');
      info.type = 'button';
      info.className = 'r8k-btn3 r8k-btn--sm';
      info.textContent = kind === 'ok' ? 'Принята' : 'Отклонена';
      info.disabled = true;
      info.style.opacity = '.45';
      info.style.cursor = 'default';
      acts.appendChild(info);
    };

    var openModal = function (row) {
      currentRow = row;
      if (m.name) m.name.textContent = row.getAttribute('data-name') || '—';
      if (m.contact) m.contact.textContent = row.getAttribute('data-contact') || '—';
      if (m.date) m.date.textContent = row.getAttribute('data-date') || '—';
      if (m.genre) m.genre.textContent = row.getAttribute('data-genre') || '—';
      if (m.note) m.note.textContent = row.getAttribute('data-note') || '—';
      var badge = row.querySelector('.r8k-badge');
      var text = badge ? badge.textContent : 'Ожидает';
      if (m.status) m.status.textContent = text;
      if (m.badge) {
        m.badge.className = 'r8k-badge ' + (badge ? badge.className.replace('r8k-badge', '').trim() : 'r8k-badge--wait');
        m.badge.textContent = text;
      }
      overlay.hidden = false;
    };
    var closeModal = function () { overlay.hidden = true; currentRow = null; };

    table.addEventListener('click', function (e) {
      var target = e.target;
      if (target.closest && target.closest('.r8k-accept')) {
        var rowA = target.closest('tr');
        if (rowA) setStatus(rowA, 'ok');
        return;
      }
      if (target.closest && target.closest('.r8k-reject')) {
        var rowR = target.closest('tr');
        if (rowR) setStatus(rowR, 'no');
        return;
      }
      var row = target.closest ? target.closest('tr[data-name]') : null;
      if (row) openModal(row);
    });

    var closeBtn = document.getElementById('r8k-close');
    if (closeBtn) closeBtn.addEventListener('click', closeModal);
    overlay.addEventListener('click', function (e) { if (e.target === overlay) closeModal(); });
    document.addEventListener('keydown', function (e) { if (e.key === 'Escape' && !overlay.hidden) closeModal(); });

    var mAccept = document.getElementById('r8k-m-accept');
    var mReject = document.getElementById('r8k-m-reject');
    if (mAccept) mAccept.addEventListener('click', function () { if (currentRow) setStatus(currentRow, 'ok'); closeModal(); });
    if (mReject) mReject.addEventListener('click', function () { if (currentRow) setStatus(currentRow, 'no'); closeModal(); });
  }

  /* Стр. 6: выбор фото и панель массового удаления */
  var gal = document.getElementById('r8k-gal');
  var bulk = document.getElementById('r8k-bulk');
  var bulkCount = document.getElementById('r8k-bulk-count');
  if (gal && bulk && bulkCount) {
    var sync = function () {
      var checked = gal.querySelectorAll('.r8k-check:checked');
      bulkCount.textContent = 'Выбрано фото: ' + checked.length;
      bulk.classList.toggle('is-on', checked.length > 0);
      each(gal.querySelectorAll('.r8k-t'), function (tile) {
        var box = tile.querySelector('.r8k-check');
        tile.classList.toggle('is-checked', !!(box && box.checked));
      });
    };
    gal.addEventListener('change', function (e) {
      if (e.target && e.target.classList && e.target.classList.contains('r8k-check')) sync();
    });
    var bulkDel = document.getElementById('r8k-bulk-del');
    if (bulkDel) bulkDel.addEventListener('click', function () {
      each(gal.querySelectorAll('.r8k-t'), function (tile) {
        var box = tile.querySelector('.r8k-check');
        if (box && box.checked && tile.parentNode) tile.parentNode.removeChild(tile);
      });
      sync();
    });
    var bulkCancel = document.getElementById('r8k-bulk-cancel');
    if (bulkCancel) bulkCancel.addEventListener('click', function () {
      each(gal.querySelectorAll('.r8k-check'), function (box) { box.checked = false; });
      sync();
    });
    sync();
  }

  /* Стр. 7: доступ к проекту, загрузка и удаление фото */
  var passGroup = document.getElementById('r8k-p-passgroup');
  each(root.querySelectorAll('[data-r8k-priv]'), function (btn) {
    btn.addEventListener('click', function () {
      each(root.querySelectorAll('[data-r8k-priv]'), function (b) { b.classList.remove('is-active'); });
      btn.classList.add('is-active');
      if (passGroup) passGroup.hidden = btn.getAttribute('data-r8k-priv') !== 'priv';
    });
  });
  var upList = document.getElementById('r8k-up');
  var drop = document.getElementById('r8k-drop');
  if (upList && drop) {
    var pool = [
      'https://static.tildacdn.com/tila3936-3931-4664-b562-346363313838/photo-1452457005517-.jpg',
      'https://static.tildacdn.com/tila3633-3430-4964-b431-353362623030/photo-1752658918430-.jpg',
      'https://static.tildacdn.com/tila3466-3932-4564-b333-393336626633/photo-1591604466107-.jpg',
      'https://static.tildacdn.com/tila6561-6538-4363-b430-313937383662/photo-1542038784456-.jpg'
    ];
    var pIndex = 0;
    var removeBtn = function (li) {
      var btn = document.createElement('button');
      btn.type = 'button';
      btn.className = 'r8k-up__x';
      btn.setAttribute('aria-label', 'Удалить фото');
      btn.innerHTML = '<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.4" stroke-linecap="round"><path d="M6 6l12 12M18 6 6 18"/></svg>';
      btn.addEventListener('click', function () {
        if (li.parentNode) li.parentNode.removeChild(li);
      });
      return btn;
    };
    upList.addEventListener('click', function (e) {
      var x = e.target.closest ? e.target.closest('.r8k-up__x') : null;
      if (x && x.parentNode) x.parentNode.parentNode.removeChild(x.parentNode);
    });
    drop.addEventListener('click', function () {
      if (upList.children.length >= 8) return;
      var li = document.createElement('li');
      li.className = 'r8k-up__t';
      var span = document.createElement('span');
      span.className = 'r8k-ph';
      var img = document.createElement('img');
      img.src = pool[pIndex % pool.length];
      img.alt = 'Загруженное фото';
      pIndex += 1;
      span.appendChild(img);
      li.appendChild(span);
      li.appendChild(removeBtn(li));
      upList.appendChild(li);
    });
  }

  /* Стр. 8: вход / регистрация */
  var authTitle = document.getElementById('r8k-auth-title');
  if (authTitle) {
    var modeBtns = root.querySelectorAll('[data-r8k-mode]');
    var submit = document.getElementById('r8k-auth-submit');
    var altText = document.getElementById('r8k-auth-alt-text');
    var altBtn = document.getElementById('r8k-auth-alt');
    var regBlocks = root.querySelectorAll('.r8k-only-reg');
    var applyMode = function (mode) {
      each(modeBtns, function (b) { b.classList.toggle('is-active', b.getAttribute('data-r8k-mode') === mode); });
      var reg = mode === 'reg';
      authTitle.textContent = reg ? 'Регистрация' : 'Вход';
      if (submit) submit.textContent = reg ? 'Зарегистрироваться' : 'Войти';
      if (altText) altText.textContent = reg ? 'Уже есть аккаунт?' : 'Ещё нет аккаунта?';
      if (altBtn) altBtn.textContent = reg ? 'Войти' : 'Регистрация';
      each(regBlocks, function (el) { el.style.display = reg ? '' : 'none'; });
    };
    each(modeBtns, function (b) {
      b.addEventListener('click', function () { applyMode(b.getAttribute('data-r8k-mode')); });
    });
    if (altBtn) altBtn.addEventListener('click', function () {
      applyMode(altBtn.textContent === 'Войти' ? 'login' : 'reg');
    });
  }

  /* Стр. 9: отправка сообщения */
  var send = document.getElementById('r8k-chat-send');
  var chatIn = document.getElementById('r8k-chat-in');
  var msgs = document.getElementById('r8k-msgs');
  if (send && chatIn && msgs) {
    var push = function () {
      var text = chatIn.value.trim();
      if (!text) return;
      var bubble = document.createElement('div');
      bubble.className = 'r8k-msg r8k-msg--out';
      bubble.textContent = text;
      var time = document.createElement('span');
      time.className = 'r8k-msg__time';
      time.textContent = 'сейчас';
      bubble.appendChild(time);
      msgs.appendChild(bubble);
      chatIn.value = '';
      msgs.scrollTop = msgs.scrollHeight;
    };
    send.addEventListener('click', push);
    chatIn.addEventListener('keydown', function (e) { if (e.key === 'Enter') push(); });
  }

  /* Стр. 10: переключатели */
  each(root.querySelectorAll('.r8k-sw'), function (sw) {
    sw.addEventListener('click', function () {
      var on = sw.classList.toggle('is-on');
      sw.setAttribute('aria-checked', on ? 'true' : 'false');
    });
  });
})();
</script>
